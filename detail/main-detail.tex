\NewDocumentCommand{\treerel}{O{} m m m}{%
  \begingroup\pgfkeys{/treerel/.cd,#1}%
  \ifx\treerelvia\treerelviaDirect
    \draw[rel arrow] (#2.\treerelfrom) -- (#3.\treerelto)
      node[rel label, midway, anchor=west, xshift=2pt] {#4};
  \else\ifx\treerelvia\treerelviaRight
    \draw[rel arrow] let \p1=(#2.east), \p2=(#3.east), \p3=(current bounding box.north east) in
      (\p1) -- (\x3+\treerelright cm, \y1) -- (\x3+\treerelright cm, \y2) -- (\p2)
      node[rel label] at (\x3+\treerelright cm, {(\y1+\y2)/2}) {#4};
  \else\ifx\treerelvia\treerelviaLeft
    \draw[rel arrow] let \p1=(#2.west), \p2=(#3.west), \p3=(current bounding box.south west) in
      (\p1) -- (\x3-\treerelright cm, \y1) -- (\x3-\treerelright cm, \y2) -- (\p2)
      node[rel label] at (\x3-\treerelright cm, {(\y1+\y2)/2}) {#4};
  \else\ifx\treerelvia\treerelviaUp
    \draw[rel arrow] let \p1=(#2.north), \p2=(#3.north), \p3=(current bounding box.north east) in
      (\p1) -- (\x1, \y3+\treerelup cm) -- (\x2, \y3+\treerelup cm) -- (\p2)
      node[rel label] at ({(\x1+\x2)/2}, \y3+\treerelup cm) {#4};
  \else\ifx\treerelvia\treerelviaDown
    \draw[rel arrow] let \p1=(#2.south), \p2=(#3.south), \p3=(current bounding box.south west) in
      (\p1) -- (\x1, \y3-\treerelup cm) -- (\x2, \y3-\treerelup cm) -- (\p2)
      node[rel label] at ({(\x1+\x2)/2}, \y3-\treerelup cm) {#4};
  \else\ifx\treerelvia\treerelviaRU
    \draw[rel arrow] let \p1=(#2.east), \p2=(#3.north), \p3=(current bounding box.north east) in
      (\p1) -- (\x3+\treerelright cm, \y1) -- (\x3+\treerelright cm, \y3+\treerelup cm)
      -- (\x2, \y3+\treerelup cm) -- (\p2)
      node[rel label] at ({(\x3+\treerelright cm+\x2)/2}, \y3+\treerelup cm) {#4};
  \else
    \draw[rel arrow] let \p1=(#2.north), \p2=(#3.east), \p3=(current bounding box.north east) in
      (\p1) -- (\x1, \y3+\treerelup cm) -- (\x3+\treerelright cm, \y3+\treerelup cm)
      -- (\x3+\treerelright cm, \y2) -- (\p2)
      node[rel label] at (\x3+\treerelright cm, {(\y3+\treerelup cm+\y2)/2}) {#4};
  \fi\fi\fi\fi\fi\fi
  \endgroup}
